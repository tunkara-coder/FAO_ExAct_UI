class EXACTApp {
    constructor() {
        this.currentUser = null;
        this.currentProject = null;
        this.isOnline = navigator.onLine;
        this.init();
    }

    async init() {
        // Check authentication
        await this.checkAuth();
        
        // Initialize service worker for offline capability
        await this.initServiceWorker();
        
        // Set up online/offline detection
        this.setupConnectivityDetection();
        
        // Load appropriate view based on auth status
        this.loadView();
    }

    async checkAuth() {
        const token = localStorage.getItem('exact_token');
        if (token) {
            try {
                // Verify token validity
                const userData = await this.verifyToken(token);
                this.currentUser = userData;
            } catch (error) {
                localStorage.removeItem('exact_token');
                this.currentUser = null;
            }
        }
    }

            name: "Field Agent",
            email: "agent@gambia.gov.gm",
            role: "field_agent",
            region: "Banjul"
        };
    }

    async initServiceWorker() {
        if ('serviceWorker' in navigator) {
            try {
                const registration = await navigator.serviceWorker.register('/service-worker.js');
                console.log('Service Worker registered successfully');
            } catch (error) {
                console.error('Service Worker registration failed:', error);
            }
        }
    }

    setupConnectivityDetection() {
        window.addEventListener('online', () => {
            this.isOnline = true;
            this.updateOnlineStatus();
            this.syncOfflineData();
        });

        window.addEventListener('offline', () => {
            this.isOnline = false;
            this.updateOnlineStatus();
        });
    }

    updateOnlineStatus() {
        const indicator = document.getElementById('offline-indicator');
        if (this.isOnline) {
            indicator.classList.remove('offline-visible');
            indicator.classList.add('offline-hidden');
        } else {
            indicator.classList.remove('offline-hidden');
            indicator.classList.add('offline-visible');
        }
    }

    async syncOfflineData() {
        // Sync any pending offline operations
        const pendingOps = JSON.parse(localStorage.getItem('pending_operations') || '[]');
        
        for (const op of pendingOps) {
            try {
                await this.submitOperation(op);
                // Remove from pending operations on success
                pendingOps.splice(pendingOps.indexOf(op), 1);
            } catch (error) {
                console.error('Failed to sync operation:', op, error);
            }
        }
        
        localStorage.setItem('pending_operations', JSON.stringify(pendingOps));
    }

    async submitOperation(operation) {
        // API call to submit the operation
        // This is a simplified example
        const response = await fetch('/api/operations', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${localStorage.getItem('exact_token')}`
            },
            body: JSON.stringify(operation)
        });
        
        if (!response.ok) throw new Error('Operation failed');
        return response.json();
    }

    loadView() {
        const appElement = document.getElementById('app');
        
        if (!this.currentUser) {
            this.showLoginView(appElement);
        } else {
            this.showDashboardView(appElement);
        }
    }

    showLoginView(container) {

            </div>

    async handleLogin(event) {
        event.preventDefault();
        
        const username = document.getElementById('username').value;
        const password = document.getElementById('password').value;
        const offlineMode = document.getElementById('offline-mode').checked;

        try {
            const userData = await this.authenticate(username, password, offlineMode);
            this.currentUser = userData;
            localStorage.setItem('exact_token', 'mock_jwt_token');
            this.loadView();
        } catch (error) {
            this.showError('Login failed: ' + error.message);
        }
    }

    renderProjects(projects) {

        const totalProjects = document.getElementById('total-projects');
        const totalReductions = document.getElementById('total-reductions');

        totalProjects.textContent = projects.length;
        
        const totalEmissions = projects

                    ${project.emissions ? `<p><strong>Net Emissions:</strong> ${project.emissions} tCO₂e</p>` : ''}

        // Simple error notification
        const errorDiv = document.createElement('div');
        errorDiv.className = 'error-notification';
        errorDiv.innerHTML = `
            <span>${message}</span>
            <button onclick="this.parentElement.remove()">×</button>
        `;
        document.body.appendChild(errorDiv);
        
        setTimeout(() => {
            if (errorDiv.parentElement) {
                errorDiv.remove();
            }
        }, 5000);
    }
